    88→    88→static void Url_ParseUrl(const char* url, struct Url* out) {
    89→    89→    if (!url || !out) return;
    90→    90→    out->host[0] = '\0';
    91→    91→    out->pathAndQuery[0] = '\0';
    92→    92→    if (url[0] == '\0') return;
    93→    93→    const char* proto_sep = "://";
    94→    94→    const char* remainder = NULL;
    95→    95→    const char* path_start = NULL;
    96→    96→    const char* proto_end = strstr(url, proto_sep);
    97→    97→    if (proto_end) {
    98→    98→        size_t host_len = proto_end - url + 3;
    99→    99→        if (host_len >= URL_HOST_MAX) host_len = URL_HOST_MAX - 1;
   100→   100→        strncpy(out->host, url, host_len);
   101→   101→        out->host[host_len] = '\0';
   102→   102→        remainder = proto_end + 3;
   103→   103→        path_start = strchr(remainder, '/');
   104→   104→        if (path_start) {
   105→   105→            size_t domain_len = path_start - remainder;
   106→   106→            if (strlen(out->host) + domain_len < URL_HOST_MAX) strncat(out->host, remainder, domain_len);
   107→   107→            size_t path_len = strlen(path_start);
   108→   108→            if (path_len >= URL_PATH_MAX) path_len = URL_PATH_MAX - 1;
   109→   109→            strncpy(out->pathAndQuery, path_start, path_len);
   110→   110→            out->pathAndQuery[path_len] = '\0';
   111→   111→        } else {
   112→   112→            if (strlen(out->host) + strlen(remainder) < URL_HOST_MAX) strcat(out->host, remainder);
   113→   113→        }
   114→   114→    } else {
   115→   115→        path_start = strchr(url, '/');
   116→   116→        if (path_start) {
   117→   117→            size_t domain_len = path_start - url;
   118→   118→            if (domain_len >= URL_HOST_MAX) domain_len = URL_HOST_MAX - 1;
   119→   119→            strncpy(out->host, url, domain_len);
   120→   120→            out->host[domain_len] = '\0';
   121→   121→            size_t path_len = strlen(path_start);
   122→   122→            if (path_len >= URL_PATH_MAX) path_len = URL_PATH_MAX - 1;
   123→   123→            strncpy(out->pathAndQuery, path_start, path_len);
   124→   124→            out->pathAndQuery[path_len] = '\0';
   125→   125→        } else {
   126→   126→            strncpy(out->host, url, URL_HOST_MAX - 1);
   127→   127→            out->host[URL_HOST_MAX - 1] = '\0';
   128→   128→        }
   129→   129→    }
   130→   130→}

   158→   158→static void CreateUrl(const char* host, const char* pathAndQuery, char* out, size_t out_size) {
   159→   159→    if (!out || out_size == 0) return;
   160→   160→    size_t host_len = host ? strlen(host) : 0;
   161→   161→    size_t path_len = pathAndQuery ? strlen(pathAndQuery) : 0;
   162→   162→    size_t total = host_len + path_len;
   163→   163→    if (total >= out_size) total = out_size - 1;
   164→   164→    if (host_len > 0) memcpy(out, host, host_len > total ? total : host_len);
   165→   165→    if (path_len > 0 && total > host_len) memcpy(out + host_len, pathAndQuery, total - host_len);
   166→   166→    out[total] = '\0';
   167→   167→}

   169→   169→static void url_encode(const char* src, char* dst, size_t dst_size) {
   170→   170→    if (!dst || dst_size == 0) return;
   171→   171→    const char* hex = "0123456789ABCDEF";
   172→   172→    size_t i = 0, j = 0;
   173→   173→    while (src[i] && j + 3 < dst_size) {
   174→   174→        unsigned char c = src[i];
   175→   175→        if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') ||
   176→   176→            c == '-' || c == '_' || c == '.' || c == '~') {
   177→   177→            dst[j++] = c;
   178→   178→        } else {
   179→   179→            dst[j++] = '%';
   180→   180→            dst[j++] = hex[c >> 4];
   181→   181→            dst[j++] = hex[c & 15];
   182→   182→        }
   183→   183→        i++;
   184→   184→    }
   185→   185→    dst[j] = '\0';
   186→   186→}

   510→   510→static int GetModuleInfo(const char* moduleName, struct ModuleInfo* outInfo) {
   511→   511→    FILE* fp = fopen("/proc/self/maps", "rt");
   512→   512→    if (!fp) return 0;
   513→   513→    char line[512];
   514→   514→    uintptr_t minStart = ~(uintptr_t)0;
   515→   515→    uintptr_t maxEnd = 0;
   516→   516→    outInfo->textSectionStart = 0;
   517→   517→    outInfo->textSectionSize = 0;
   518→   518→    int found = 0;
   519→   519→    while (fgets(line, sizeof(line), fp)) {
   520→   520→        if (strstr(line, moduleName)) {
   521→   521→            uintptr_t start, end;
   522→   522→            char perms[5];
   523→   523→            if (sscanf(line, "%lx-%lx %4s", &start, &end, perms) == 3) {
   524→   524→                found = 1;
   525→   525→                if (start < minStart) minStart = start;
   526→   526→                if (end > maxEnd) maxEnd = end;
   527→   527→                if (perms[0] == 'r' && perms[2] == 'x') {
   528→   528→                    if (outInfo->textSectionStart == 0) outInfo->textSectionStart = start;
   529→   529→                    outInfo->textSectionSize += (end - start);
   530→   530→                }
   531→   531→            }
   532→   532→        }
   533→   533→    }
   534→   534→    fclose(fp);
   535→   535→    if (found) {
   536→   536→        outInfo->baseAddress = minStart;
   537→   537→        outInfo->size = maxEnd - minStart;
   538→   538→    }
   539→   539→    return found;
   540→   540→}

   542→   542→static uintptr_t FindStringInMemory(uintptr_t start, uintptr_t size, const char* target, int wide) {
   543→   543→    if (!wide) {
   544→   544→        size_t targetLen = strlen(target);
   545→   545→        uintptr_t maxAddr = start + size - targetLen;
   546→   546→        uintptr_t addr;
   547→   547→        for (addr = start; addr < maxAddr; addr++) {
   548→   548→            if (memcmp((void*)addr, target, targetLen) == 0) return addr;
   549→   549→        }
   550→   550→    } else {
   551→   551→        size_t targetLen = strlen(target);
   552→   552→        size_t wideLen = targetLen * 2;
   553→   553→        uintptr_t maxAddr = start + size - wideLen;
   554→   554→        uintptr_t addr;
   555→   555→        for (addr = start; addr < maxAddr; addr++) {
   556→   556→            int match = 1;
   557→   557→            size_t i;
   558→   558→            for (i = 0; i < targetLen; i++) {
   559→   559→                if (((uint8_t*)addr)[i * 2] != target[i] || ((uint8_t*)addr)[i * 2 + 1] != 0) {
   560→   560→                    match = 0;
   561→   561→                    break;
   562→   562→                }
   563→   563→            }
   564→   564→            if (match) return addr;
   565→   565→        }
   566→   566→    }
   567→   567→    return 0;
   568→   568→}
