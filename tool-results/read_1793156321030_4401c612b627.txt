    27→static char* FString_ToCStr(const struct FString* str) {
    28→    if (!str || !str->data || str->length <= 0) return NULL;
    29→    int maxBytes = (str->length * 3) + 1;
    30→    char* out = (char*)malloc(maxBytes);
    31→    if (!out) return NULL;
    32→    int o = 0;
    33→    int i;
    34→    for (i = 0; i < str->length - 1; i++) {
    35→        uint16_t c = str->data[i];
    36→        if (c <= 0x7F) { out[o++] = (char)c; }
    37→        else if (c <= 0x7FF) { out[o++] = (char)(0xC0 | (c >> 6)); out[o++] = (char)(0x80 | (c & 0x3F)); }
    38→        else { out[o++] = (char)(0xE0 | (c >> 12)); out[o++] = (char)(0x80 | ((c >> 6) & 0x3F)); out[o++] = (char)(0x80 | (c & 0x3F)); }
    39→    }
    40→    out[o] = 0;
    41→    return out;
    42→}

    44→static struct FString FString_FromCStr(const char* str) {
    45→    struct FString fs;
    46→    memset(&fs, 0, sizeof(fs));
    47→    if (!str) return fs;
    48→    int utf16len = 0;
    49→    const unsigned char* s = (const unsigned char*)str;
    50→    while (*s) {
    51→        if ((*s & 0x80) == 0) { s += 1; utf16len += 1; }
    52→        else if ((*s & 0xE0) == 0xC0) { s += 2; utf16len += 1; }
    53→        else if ((*s & 0xF0) == 0xE0) { s += 3; utf16len += 1; }
    54→        else { s += 4; utf16len += 1; }
    55→    }
    56→    fs.length = fs.capacity = utf16len + 1;
    57→    fs.data = (TCHAR*)malloc(fs.capacity * sizeof(TCHAR));
    58→    if (!fs.data) { fs.length = fs.capacity = 0; return fs; }
    59→    int i = 0;
    60→    s = (const unsigned char*)str;
    61→    while (*s) {
    62→        if ((*s & 0x80) == 0) { fs.data[i++] = *s++; }
    63→        else if ((*s & 0xE0) == 0xC0) { fs.data[i++] = (uint16_t)(((s[0] & 0x1F) << 6) | (s[1] & 0x3F)); s += 2; }
    64→        else if ((*s & 0xF0) == 0xE0) { fs.data[i++] = (uint16_t)(((s[0] & 0x0F) << 12) | ((s[1] & 0x3F) << 6) | (s[2] & 0x3F)); s += 3; }
    65→        else { fs.data[i++] = '?'; s += 4; }
    66→    }
    67→    fs.data[i] = 0;
    68→    return fs;
    69→}

    88→static void Url_ParseUrl(const char* url, struct Url* out) {
    89→    if (!url || !out) return;
    90→    out->host[0] = '\0';
    91→    out->pathAndQuery[0] = '\0';
    92→    if (url[0] == '\0') return;
    93→    const char* proto_sep = "://";
    94→    const char* remainder = NULL;
    95→    const char* path_start = NULL;
    96→    const char* proto_end = strstr(url, proto_sep);
    97→    if (proto_end) {
    98→        size_t host_len = proto_end - url + 3;
    99→        if (host_len >= URL_HOST_MAX) host_len = URL_HOST_MAX - 1;
   100→        strncpy(out->host, url, host_len);
   101→        out->host[host_len] = '\0';
   102→        remainder = proto_end + 3;
   103→        path_start = strchr(remainder, '/');
   104→        if (path_start) {
   105→            size_t domain_len = path_start - remainder;
   106→            if (strlen(out->host) + domain_len < URL_HOST_MAX) strncat(out->host, remainder, domain_len);
   107→            size_t path_len = strlen(path_start);
   108→            if (path_len >= URL_PATH_MAX) path_len = URL_PATH_MAX - 1;
   109→            strncpy(out->pathAndQuery, path_start, path_len);
   110→            out->pathAndQuery[path_len] = '\0';
   111→        } else {
   112→            if (strlen(out->host) + strlen(remainder) < URL_HOST_MAX) strcat(out->host, remainder);
   113→        }
   114→    } else {
   115→        path_start = strchr(url, '/');
   116→        if (path_start) {
   117→            size_t domain_len = path_start - url;
   118→            if (domain_len >= URL_HOST_MAX) domain_len = URL_HOST_MAX - 1;
   119→            strncpy(out->host, url, domain_len);
   120→            out->host[domain_len] = '\0';
   121→            size_t path_len = strlen(path_start);
   122→            if (path_len >= URL_PATH_MAX) path_len = URL_PATH_MAX - 1;
   123→            strncpy(out->pathAndQuery, path_start, path_len);
   124→            out->pathAndQuery[path_len] = '\0';
   125→        } else {
   126→            strncpy(out->host, url, URL_HOST_MAX - 1);
   127→            out->host[URL_HOST_MAX - 1] = '\0';
   128→        }
   129→    }
   130→}

   132→static int ShouldRedirect(const char* host) {
   133→    char stripped[URL_HOST_MAX];
   134→    strncpy(stripped, host, sizeof(stripped) - 1);
   135→    stripped[sizeof(stripped) - 1] = '\0';
   136→    if (strncmp(stripped, "http://", 7) == 0) memmove(stripped, stripped + 7, strlen(stripped + 7) + 1);
   137→    else if (strncmp(stripped, "https://", 8) == 0) memmove(stripped, stripped + 8, strlen(stripped + 8) + 1);
   138→    char* colon = strchr(stripped, ':');
   139→    if (colon) *colon = '\0';
   140→    const char* domains[] = {
   141→        "game-social.epicgames.com",
   142→        "ol.epicgames.com",
   143→        "ol.epicgames.net",
   144→        "on.epicgames.com",
   145→        "ak.epicgames.com",
   146→        "epicgames.dev"
   147→    };
   148→    int num_domains = sizeof(domains) / sizeof(domains[0]);
   149→    int i;
   150→    for (i = 0; i < num_domains; i++) {
   151→        size_t dlen = strlen(domains[i]);
   152→        size_t slen = strlen(stripped);
   153→        if (slen >= dlen && strcmp(stripped + slen - dlen, domains[i]) == 0) return 1;
   154→    }
   155→    return 0;
   156→}

   158→static void CreateUrl(const char* host, const char* pathAndQuery, char* out, size_t out_size) {
   159→    if (!out || out_size == 0) return;
   160→    size_t host_len = host ? strlen(host) : 0;
   161→    size_t path_len = pathAndQuery ? strlen(pathAndQuery) : 0;
   162→    size_t total = host_len + path_len;
   163→    if (total >= out_size) total = out_size - 1;
   164→    if (host_len > 0) memcpy(out, host, host_len > total ? total : host_len);
   165→    if (path_len > 0 && total > host_len) memcpy(out + host_len, pathAndQuery, total - host_len);
   166→    out[total] = '\0';
   167→}

   169→static void url_encode(const char* src, char* dst, size_t dst_size) {
   170→    if (!dst || dst_size == 0) return;
   171→    const char* hex = "0123456789ABCDEF";
   172→    size_t i = 0, j = 0;
   173→    while (src[i] && j + 3 < dst_size) {
   174→        unsigned char c = src[i];
   175→        if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') ||
   176→            c == '-' || c == '_' || c == '.' || c == '~') {
   177→            dst[j++] = c;
   178→        } else {
   179→            dst[j++] = '%';
   180→            dst[j++] = hex[c >> 4];
   181→            dst[j++] = hex[c & 15];
   182→        }
   183→        i++;
   184→    }
   185→    dst[j] = '\0';
   186→}

   188→static void send_version_report(const char* version) {
   189→    struct Url url;
   190→    Url_ParseUrl(g_backend_url, &url);
   191→    char host[256];
   192→    strncpy(host, url.host, sizeof(host)-1);
   193→    if (strncmp(host, "http://", 7) == 0) memmove(host, host+7, strlen(host+7)+1);
   194→    else if (strncmp(host, "https://", 8) == 0) memmove(host, host+8, strlen(host+8)+1);
   195→    char* port_str = "80";
   196→    char* colon = strchr(host, ':');
   197→    if (colon) {
   198→        *colon = '\0';
   199→        port_str = colon+1;
   200→    }
   201→    struct addrinfo hints, *res;
   202→    memset(&hints, 0, sizeof(hints));
   203→    hints.ai_family = AF_UNSPEC;
   204→    hints.ai_socktype = SOCK_STREAM;
   205→    int err = getaddrinfo(host, port_str, &hints, &res);
   206→    if (err) return;
   207→    int sock = socket(res->ai_family, res->ai_socktype, res->ai_protocol);
   208→    if (sock < 0) { freeaddrinfo(res); return; }
   209→    if (connect(sock, res->ai_addr, res->ai_addrlen) < 0) { close(sock); freeaddrinfo(res); return; }
   210→    char body[512];
   211→    snprintf(body, sizeof(body), "GameVersion = \"%s\"", version);
   212→    char request[1024];
   213→    snprintf(request, sizeof(request),
   214→        "POST / HTTP/1.1\r\n"
   215→        "Host: %s\r\n"
   216→        "Content-Type: text/plain\r\n"
   217→        "Content-Length: %zu\r\n"
   218→        "Connection: close\r\n"
   219→        "\r\n"
   220→        "%s", host, strlen(body), body);
   221→    send(sock, request, strlen(request), 0);
   222→    close(sock);
   223→    freeaddrinfo(res);
   224→}

   510→static int GetModuleInfo(const char* moduleName, struct ModuleInfo* outInfo) {
   511→    FILE* fp = fopen("/proc/self/maps", "rt");
   512→    if (!fp) return 0;
   513→    char line[512];
   514→    uintptr_t minStart = ~(uintptr_t)0;
   515→    uintptr_t maxEnd = 0;
   516→    outInfo->textSectionStart = 0;
   517→    outInfo->textSectionSize = 0;
   518→    int found = 0;
   519→    while (fgets(line, sizeof(line), fp)) {
   520→        if (strstr(line, moduleName)) {
   521→            uintptr_t start, end;
   522→            char perms[5];
   523→            if (sscanf(line, "%lx-%lx %4s", &start, &end, perms) == 3) {
   524→                found = 1;
   525→                if (start < minStart) minStart = start;
   526→                if (end > maxEnd) maxEnd = end;
   527→                if (perms[0] == 'r' && perms[2] == 'x') {
   528→                    if (outInfo->textSectionStart == 0) outInfo->textSectionStart = start;
   529→                    outInfo->textSectionSize += (end - start);
   530→                }
   531→            }
   532→        }
   533→    }
   534→    fclose(fp);
   535→    if (found) {
   536→        outInfo->baseAddress = minStart;
   537→        outInfo->size = maxEnd - minStart;
   538→    }
   539→    return found;
   540→}

   542→static uintptr_t FindStringInMemory(uintptr_t start, uintptr_t size, const char* target, int wide) {
   543→    if (!wide) {
   544→        size_t targetLen = strlen(target);
   545→        uintptr_t maxAddr = start + size - targetLen;
   546→        uintptr_t addr;
   547→        for (addr = start; addr < maxAddr; addr++) {
   548→            if (memcmp((void*)addr, target, targetLen) == 0) return addr;
   549→        }
   550→    } else {
   551→        size_t targetLen = strlen(target);
   552→        size_t wideLen = targetLen * 2;
   553→        uintptr_t maxAddr = start + size - wideLen;
   554→        uintptr_t addr;
   555→        for (addr = start; addr < maxAddr; addr++) {
   556→            int match = 1;
   557→            size_t i;
   558→            for (i = 0; i < targetLen; i++) {
   559→                if (((uint8_t*)addr)[i * 2] != target[i] || ((uint8_t*)addr)[i * 2 + 1] != 0) {
   560→                    match = 0;
   561→                    break;
   562→                }
   563→            }
   564→            if (match) return addr;
   565→        }
   566→    }
   567→    return 0;
   568→}

   783→static void clear_cache_arm64(void *start, void *end) {
   784→  uint64_t xstart = (uint64_t)(uintptr_t)start;
   785→  uint64_t xend = (uint64_t)(uintptr_t)end;
   786→  static uint64_t ctr_el0 = 0;
   787→  if (ctr_el0 == 0) {
   788→    __asm__ __volatile__("mrs %0, ctr_el0" : "=r"(ctr_el0));
   789→  }
   790→  uint64_t addr;
   791→  if (((ctr_el0 >> 28) & 0x1) == 0x0) {
   792→    const size_t dcache_line_size = 4 << ((ctr_el0 >> 16) & 15);
   793→    for (addr = xstart & ~(dcache_line_size - 1); addr < xend; addr += dcache_line_size)
   794→      __asm__ __volatile__("dc cvau, %0" ::"r"(addr));
   795→  }
   796→  __asm__ __volatile__("dsb ish");
   797→  if (((ctr_el0 >> 29) & 0x1) == 0x0) {
   798→    const size_t icache_line_size = 4 << ((ctr_el0 >> 0) & 15);
   799→    for (addr = xstart & ~(icache_line_size - 1); addr < xend; addr += icache_line_size)
   800→      __asm__ __volatile__("ic ivau, %0" ::"r"(addr));
   801→    __asm__ __volatile__("dsb ish");
   802→  }
   803→  __asm__ __volatile__("isb sy");
   804→}

   883→static MemBlock alloc_mem_block(size_t in_size, int is_exec) {
   884→  if (in_size > (size_t)os_page_size()) { MemBlock b; MemBlock_init(&b, 0, 0); return b; }
   885→  uint8_t *result = NULL;
   886→  int count;
   887→  simple_linear_allocator_t **allocators;
   888→  int *count_ptr;
   889→  if (is_exec) { count = gMemoryAllocator.code_page_count; allocators = gMemoryAllocator.code_page_allocators; count_ptr = &gMemoryAllocator.code_page_count; }
   890→  else { count = gMemoryAllocator.data_page_count; allocators = gMemoryAllocator.data_page_allocators; count_ptr = &gMemoryAllocator.data_page_count; }
   891→  for (int i = 0; i < count; i++) { result = (uint8_t *)simple_linear_allocator_alloc(allocators[i], (uint32_t)in_size, 0); if (result) break; }
   892→  if (!result) {
   893→    if (*count_ptr >= MAX_PAGE_ALLOCATORS) { MemBlock b; MemBlock_init(&b, 0, 0); return b; }
   894→    void *page = os_memory_allocate(os_page_size(), kNoAccess);
   895→    if (!page) { MemBlock b; MemBlock_init(&b, 0, 0); return b; }
   896→    os_memory_set_permission(page, os_page_size(), is_exec ? kReadExecute : kReadWrite);
   897→    simple_linear_allocator_t *pa = (simple_linear_allocator_t *)malloc(sizeof(simple_linear_allocator_t));
   898→    simple_linear_allocator_init(pa, (uint8_t *)page, os_page_size(), 8);
   899→    allocators[*count_ptr] = pa; (*count_ptr)++;
   900→    result = (uint8_t *)simple_linear_allocator_alloc(pa, (uint32_t)in_size, 0);
   901→  }
   902→  MemBlock b; MemBlock_init(&b, (addr_t)result, in_size); return b;
   903→}

   908→static int DobbyCodePatch(void *address, uint8_t *buffer, uint32_t buffer_size) {
   909→  int page_size = os_page_size();
   910→  uintptr_t patch_page = ALIGN_FLOOR(address, page_size);

   912→  mprotect((void *)patch_page, page_size, PROT_READ | PROT_WRITE | PROT_EXEC);
   913→  if (patch_page != patch_end_page) mprotect((void *)patch_end_page, page_size, PROT_READ | PROT_WRITE | PROT_EXEC);
   914→  memcpy(address, buffer, buffer_size);
   915→  mprotect((void *)patch_page, page_size, PROT_READ | PROT_EXEC);
   916→  if (patch_page != patch_end_page) mprotect((void *)patch_end_page, page_size, PROT_READ | PROT_EXEC);
   917→  addr_t clear_start = (addr_t)address;
   918→  clear_cache_arm64((void *)clear_start, (void *)(clear_start + buffer_size));

  1053→static int get_memory_layout(MemRegionInfo *out_regions, int max_regions) {
  1054→  FILE *fp = fopen("/proc/self/maps", "r");
  1055→  if (!fp) return 0;
  1056→  int count = 0;
  1057→  char line[2048];
  1058→  while (fgets(line, sizeof(line), fp) && count < max_regions) {
  1059→    addr_t region_start, region_end, region_offset;
  1060→    char permissions[5] = {0};
  1061→    uint8_t dev_major, dev_minor;
  1062→    long inode;
  1063→    int path_index = 0;
  1064→    if (sscanf(line, "%lx-%lx %4c %lx %hhx:%hhx %ld %n", (unsigned long *)&region_start, (unsigned long *)&region_end, permissions, (unsigned long *)&region_offset, &dev_major, &dev_minor, &inode, &path_index) < 7) continue;
  1065→    int perm = 0;
  1066→    if (permissions[0] == 'r') perm |= MEM_PERM_R;
  1067→    if (permissions[1] == 'w') perm |= MEM_PERM_W;
  1068→    if (permissions[2] == 'x') perm |= MEM_PERM_X;
  1069→    out_regions[count].addr = region_start;
  1070→    out_regions[count].size = region_end - region_start;
  1071→    out_regions[count].perm = perm;
  1072→    count++;
  1073→  }
  1074→  fclose(fp);
  1075→  return count;
  1076→}

  1078→static MemBlock alloc_near_block_in_page(addr_t pos, size_t range, uint32_t in_size) {
  1079→  if (custom_alloc_near_code_handler) { addr_t a = custom_alloc_near_code_handler(in_size, pos, range); if (a) { MemBlock b; MemBlock_init(&b, a, in_size); return b; } }
  1080→  MemRange search_range;
  1081→  MemRange_init(&search_range, pos > range ? pos - range : 0, range * 2);
  1082→  for (int i = 0; i < gMemoryAllocator.code_page_count; i++) {
  1083→    simple_linear_allocator_t *al = gMemoryAllocator.code_page_allocators[i];
  1084→    addr_t cursor = (addr_t)al->buffer + al->size;
  1085→    uint32_t unused_size = al->capacity - al->size;
  1086→    MemRange unused_range; MemRange_init(&unused_range, cursor, unused_size);
  1087→    MemRange intersect = MemRange_intersect(&search_range, &unused_range);
  1088→    if (intersect.size < in_size) continue;
  1089→    addr_t gap_size = intersect.start_ - cursor;
  1090→    if (gap_size) simple_linear_allocator_alloc(al, (uint32_t)gap_size, 0);
  1091→    uint8_t *result = simple_linear_allocator_alloc(al, in_size, 0);
  1092→    MemBlock b; MemBlock_init(&b, (addr_t)result, in_size); return b;
  1093→  }
  1094→  MemRegionInfo regions[MAX_REGIONS];
  1095→  int region_count = get_memory_layout(regions, MAX_REGIONS);
  1096→  for (int i = 0; i < region_count; i++) {
  1097→    if (i >= region_count - 1) break;
  1098→    addr_t unused_start = regions[i].addr + regions[i].size;
  1099→    size_t unused_size = regions[i + 1].addr - unused_start;
  1100→    MemRange unused_range; MemRange_init(&unused_range, unused_start, unused_size);
  1101→    MemRange intersect = MemRange_intersect(&search_range, &unused_range);
  1102→    if (intersect.size < in_size) continue;
  1103→    void *unused_page = (void *)ALIGN_FLOOR(intersect.start_, os_page_size());
  1104→    void *page = mmap(unused_page, os_page_size(), PROT_READ | PROT_EXEC, MAP_PRIVATE | MAP_ANONYMOUS | MAP_FIXED, -1, 0);
  1105→    if (page == MAP_FAILED) continue;
  1106→    if (page != unused_page) continue;
  1107→    if (gMemoryAllocator.code_page_count >= MAX_PAGE_ALLOCATORS) break;
  1108→    simple_linear_allocator_t *pa = (simple_linear_allocator_t *)malloc(sizeof(simple_linear_allocator_t));
  1109→    simple_linear_allocator_init(pa, (uint8_t *)page, os_page_size(), 8);
  1110→    gMemoryAllocator.code_page_allocators[gMemoryAllocator.code_page_count++] = pa;
  1111→    return alloc_near_block_in_page(pos, range, in_size);
  1112→  }
  1113→  const uint8_t zero_seq[0x1000] = {0};
  1114→  for (int i = 0; i < region_count; i++) {
  1115→    if (!(regions[i].perm & MEM_PERM_X)) continue;
  1116→    MemRange r; MemRange_init(&r, regions[i].addr, regions[i].size);
  1117→    MemRange intersect = MemRange_intersect(&search_range, &r);
  1118→    if (intersect.size < in_size) continue;
  1119→    const uint8_t *p = (const uint8_t *)intersect.start_;
  1120→    size_t search_size = intersect.size;
  1121→    size_t needle = in_size + 3;
  1122→    void *found = NULL;
  1123→    for (size_t j = 0; j + needle <= search_size; j++) { if (memcmp(p + j, zero_seq, needle) == 0) { found = (void *)(p + j); break; } }
  1124→    if (!found) continue;
  1125→    found = (void *)ALIGN_CEIL(found, 4);
  1126→    MemBlock b; MemBlock_init(&b, (addr_t)found, in_size); return b;
  1127→  }
  1128→  MemBlock b; MemBlock_init(&b, 0, 0); return b;
  1129→}

  1196→static int relocate(ReloCtx *ctx, int branch) {
  1197→  Assembler *as = &ctx->as;
  1198→  uint32_t offset = 0;
  1199→  while (offset < ctx->origin_size) {
  1200→    uint32_t inst = *(uint32_t *)(origin_cursor(ctx, offset));
  1201→    if (inst_is_b_bl(inst)) {
  1202→      int64_t off = decode_imm26_offset(inst);
  1203→      addr_t dst = origin_cursor(ctx, offset) + off;
  1204→      RelocDataLabel *label = Assembler_create_data_label(as, (uint64_t)dst, 8);
  1205→      Assembler_ldr_label(as, TMP_REG_0, label);
  1206→      if ((inst & UnconditionalBranchMask) == BL) Assembler_blr(as, TMP_REG_0);
  1207→      else Assembler_br(as, TMP_REG_0);
  1208→    } else if (inst_is_ldr_literal(inst)) {
  1209→      int64_t off = decode_imm19_offset(inst);
  1210→      addr_t dst = origin_cursor(ctx, offset) + off;
  1211→      int rt = decode_rt(inst);
  1212→      char opc = bits_(inst, 30, 31);
  1213→      Assembler_mov_imm64(as, TMP_REG_0, (uint64_t)dst);
  1214→      if (opc == 0b00) Assembler_ldr_unsigned_w(as, rt, TMP_REG_0, 0);
  1215→      else if (opc == 0b01) Assembler_ldr_unsigned_x(as, rt, TMP_REG_0, 0);
  1216→    } else if (inst_is_adr(inst)) {
  1217→      int64_t off = decode_immhi_immlo_offset(inst);
  1218→      addr_t dst = origin_cursor(ctx, offset) + off;
  1219→      int rd = decode_rd(inst);
  1220→      Assembler_mov_imm64(as, rd, (uint64_t)dst);
  1221→    } else if (inst_is_adrp(inst)) {
  1222→      int64_t off = decode_immhi_immlo_zero12_offset(inst);
  1223→      addr_t dst = origin_cursor(ctx, offset) + off;
  1224→      dst = ALIGN(dst, 0x1000);
  1225→      int rd = decode_rd(inst);
  1226→      Assembler_mov_imm64(as, rd, (uint64_t)dst);
  1227→    } else if (inst_is_b_cond(inst)) {
  1228→      int64_t off = decode_imm19_offset(inst);
  1229→      addr_t dst = origin_cursor(ctx, offset) + off;
  1230→      uint32_t branch_inst = inst;
  1231→      char cond = bits_(inst, 0, 3);
  1232→      cond = cond ^ 1;
  1233→      set_bits_(branch_inst, 0, 3, cond);
  1234→      int64_t new_off = 4 * 3;
  1235→      uint32_t imm19 = (uint32_t)(new_off >> 2);
  1236→      set_bits_(branch_inst, 5, 23, imm19);
  1237→      RelocDataLabel *label = Assembler_create_data_label(as, (uint64_t)dst, 8);
  1238→      Assembler_emit_u32(as, branch_inst);
  1239→      Assembler_ldr_label(as, TMP_REG_0, label);
  1240→      Assembler_br(as, TMP_REG_0);
  1241→    } else if (inst_is_compare_b(inst)) {
  1242→      int64_t off = decode_imm19_offset(inst);
  1243→      addr_t dst = origin_cursor(ctx, offset) + off;
  1244→      uint32_t branch_inst = inst;
  1245→      char op = bit_(inst, 24);
  1246→      op = op ^ 1;
  1247→      set_bit_(branch_inst, 24, op);
  1248→      int64_t new_off = 4 * 3;
  1249→      uint32_t imm19 = (uint32_t)(new_off >> 2);
  1250→      set_bits_(branch_inst, 5, 23, imm19);
  1251→      RelocDataLabel *label = Assembler_create_data_label(as, (uint64_t)dst, 8);
  1252→      Assembler_emit_u32(as, branch_inst);
  1253→      Assembler_ldr_label(as, TMP_REG_0, label);
  1254→      Assembler_br(as, TMP_REG_0);
  1255→    } else if (inst_is_test_b(inst)) {
  1256→      int64_t off = decode_imm14_offset(inst);
  1257→      addr_t dst = origin_cursor(ctx, offset) + off;
  1258→      uint32_t branch_inst = inst;
  1259→      char op = bit_(inst, 24);
  1260→      op = op ^ 1;
  1261→      set_bit_(branch_inst, 24, op);
  1262→      int64_t new_off = 4 * 3;
  1263→      uint32_t imm14 = (uint32_t)(new_off >> 2);
  1264→      set_bits_(branch_inst, 5, 18, imm14);
  1265→      RelocDataLabel *label = Assembler_create_data_label(as, (uint64_t)dst, 8);
  1266→      Assembler_emit_u32(as, branch_inst);
  1267→      Assembler_ldr_label(as, TMP_REG_0, label);
  1268→      Assembler_br(as, TMP_REG_0);
  1269→    } else {
  1270→      Assembler_emit_u32(as, inst);
  1271→    }
  1272→    offset += sizeof(uint32_t);
  1273→  }
  1274→  ctx->origin_size = offset;
  1275→  if (branch) Assembler_literal_ldr_branch(as, origin_cursor(ctx, offset));
  1276→  Assembler_reloc_data_labels(as);
  1277→  ctx->relocated = Assembler_finalize(as);
  1278→  ctx->relocated_built = 1;
  1279→  return 0;
  1280→}

  1282→int DobbyHook(void *address, void *fake_func, void **out_origin_func) {
  1283→  if (!address) return -1;
  1284→  make_memory_readable(address, 4);
  1285→  if (find_entry((addr_t)address)) return -1;
  1286→  if (g_entries_count >= MAX_INTERCEPT_ENTRIES) return -1;
  1287→  InterceptEntry *entry = &g_entries[g_entries_count];
  1288→  memset(entry, 0, sizeof(*entry));
  1289→  entry->addr = (addr_t)address;
  1290→  entry->fake_func_addr = (addr_t)fake_func;
  1291→  addr_t from = (addr_t)address;
  1292→  addr_t to = (addr_t)fake_func;
  1293→  Trampoline *trampoline = NULL;
  1294→  Trampoline *near_trampoline = NULL;
  1295→  if (g_enable_near_trampoline) near_trampoline = generate_near_trampoline(from, to);
  1296→  if (!near_trampoline) trampoline = generate_normal_trampoline(from, to);
  1297→  addr_t tramp_addr; size_t tramp_size;
  1298→  if (near_trampoline) { tramp_addr = Trampoline_addr(near_trampoline); tramp_size = Trampoline_size(near_trampoline); }
  1299→  else if (trampoline) { tramp_addr = Trampoline_addr(trampoline); tramp_size = Trampoline_size(trampoline); }
  1300→  else return -1;
  1301→  ReloCtx ctx; ReloCtx_init(&ctx, from, (uint32_t)tramp_size);
  1302→  relocate(&ctx, 1);
  1303→  if (ctx.relocated.size == 0) { ReloCtx_destroy(&ctx); if (trampoline) free(trampoline); if (near_trampoline) free(near_trampoline); return -1; }
  1304→  entry->patched.start_ = from;
  1305→  entry->patched.size = (size_t)ctx.origin_size;
  1306→  entry->relocated = ctx.relocated;
  1307→  backup_orig_code(entry);
  1308→  DobbyCodePatch((void *)from, (uint8_t *)tramp_addr, (uint32_t)tramp_size);
  1309→  if (out_origin_func) *out_origin_func = (void *)entry->relocated.start_;
  1310→  g_entries_count++;
  1311→  ReloCtx_destroy(&ctx);
  1312→  if (trampoline) { free(trampoline); }
  1313→  if (near_trampoline) { free(near_trampoline); }
  1314→  return 0;
  1315→}

  1346→static int InternalProcessRequest(void* Req, int isEOS) {
  1347→    if (!Req) return 0;
  1348→    int (*OGProcessRequest)(void*) = isEOS ? EOSFCurlHook_OGProcessRequest : FCurlHook_OGProcessRequest;
  1349→    void (*SetUrl)(void*, const void*) = isEOS ? EOSFCurlHook_SetUrl : FCurlHook_SetUrl;
  1350→    struct FString* OGUrl = isEOS ? EOSFCurlHook_GetUrl(Req) : FCurlHook_GetUrl(Req);
  1351→    if (OGUrl->length == 0 || !OGUrl->data) return OGProcessRequest(Req);
  1352→    char* OGUrlCStr = FString_ToCStr(OGUrl);
  1353→    if (!OGUrlCStr) return OGProcessRequest(Req);
  1354→    struct Url url;
  1355→    Url_ParseUrl(OGUrlCStr, &url);
  1356→    if (ShouldRedirect(url.host)) {
  1357→        char newUrlCStr[URL_TOTAL_MAX];
  1358→        CreateUrl(g_backend_url, url.pathAndQuery, newUrlCStr, sizeof(newUrlCStr));
  1359→        struct FString* newUrl = (struct FString*)malloc(sizeof(struct FString));
  1360→        *newUrl = FString_FromCStr(newUrlCStr);
  1361→        SetUrl(Req, newUrl);
  1362→    }
  1363→    free(OGUrlCStr);
  1364→    return OGProcessRequest(Req);
  1365→}

  1370→static int CurlHook_SetOptHook(void* handle, int option, void* args) {
  1371→    if (!CurlHook_OG_SETOPT) return -1;
  1372→    if (option == 10002 && args != NULL) {
  1373→        const char* url = (const char*)args;
  1374→        if (url) {
  1375→            struct Url parsed;
  1376→            Url_ParseUrl(url, &parsed);
  1377→            if (ShouldRedirect(parsed.host)) {
  1378→                static __thread char newUrl[URL_TOTAL_MAX];
  1379→                CreateUrl(g_backend_url, parsed.pathAndQuery, newUrl, sizeof(newUrl));
  1380→                return CurlHook_OG_SETOPT(handle, option, (void*)newUrl);
  1381→            }
  1382→        }
  1383→    }
  1384→    return CurlHook_OG_SETOPT(handle, option, args);
  1385→}

  1404→    const char* lib_name = NULL;

  1415→            return;

  1420→        void* handle = dlopen("libUE4.so", RTLD_NOW | RTLD_NOLOAD);
  1421→        if (handle) {

  1423→            if (curl_setopt) {
  1424→                DobbyHook(curl_setopt, (void*)CurlHook_SetOptHook, (void**)&CurlHook_OG_SETOPT);
  1425→                dlclose(handle);
  1426→                return;
  1427→            }
  1428→            dlclose(handle);
  1429→        }
  1430→    }
  1431→    struct ModuleInfo mod;
  1432→    if (!GetModuleInfo(lib_name, &mod)) {
  1433→        return;
  1434→    }

  1437→        found = detect_version_by_string((const uint8_t*)mod.baseAddress, mod.size, 1);
  1438→        if (!found) found = detect_version_by_string((const uint8_t*)mod.baseAddress, mod.size, 0);
  1439→    } else {
  1440→        found = detect_version_by_string((const uint8_t*)mod.baseAddress, mod.size, 0);
  1441→    }

  1443→        return;

  1445→    send_version_report(detected_version);
  1446→    load_eos_offsets_for_version();

  1450→        void* eosBase = GetLibBase("libEOSSDK.so");
